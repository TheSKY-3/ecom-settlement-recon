> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 无 AI 模型全量判卷：引擎读脏三本账 + 底座库 → 六类发现清单（含证据/置信度/审计链）→ 对照标准答案出分 → 成绩单入库 → 结果接进工作台。

**Architecture:** 四个纯标准库模块（audit/matcher/classifier/verifier）+ 判分器（evals/score.py）+ 一个入口脚本；金额期望**复用** `src/synth/bills.py` 规则卡实现（同一口径）。规则树与判分口径见 spec 第 1、4 节。

**Tech Stack:** 纯标准库（csv/sqlite3/hashlib/json）；判分与工作台复用已有 FastAPI 层。

**Spec:** `docs/superpowers/specs/2026-09-26-p2-engine-design.md`

## Global Constraints

- 引擎不改任何数据（只读）；发现结构 = gt 六字段 + confidence/description/evidence/verified。
- 变体名与 P1 标准答案逐字一致：删账单/孤儿账单/删支付/费率错误/漏运费/取消单被开账单/退款缺失/整单重复/两版并存。
- 容差 ±0.005（`src/synth/config.py: TOLERANCE`）；期望金额 = `amounts_from_items()`（`src/synth/bills.py`）。
- 一单至多一条金额类发现：先形态（孤儿/状态/版本/重复/单边）后金额（账单侧→支付侧→退款侧）。
- 引擎产物 `output/engine/`（gitignore）；成绩单 `docs/reports/p2-engine-report.md`（入库）。
- 测试不依赖真实数据：`RECON_SYNTHETIC_DIR` / `RECON_DB_PATH` / `RECON_ENGINE_DIR` 环境变量。

"""审计链：每步记录输入/输出指纹并环环相扣（哈希含上一步哈希），篡改任何一环即断链。"""
from __future__ import annotations

import hashlib
import json
from pathlib import Path

ZERO = "0" * 64


def sha256_of(path: Path) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def _digest(payload: dict, prev_hash: str) -> str:
    blob = json.dumps({**payload, "prev_hash": prev_hash}, sort_keys=True, ensure_ascii=False)
    return hashlib.sha256(blob.encode("utf-8")).hexdigest()


class AuditChain:
    def __init__(self, run_id: str):
        self.run_id = run_id
        self.steps: list[dict] = []

    def add_step(self, action: str, inputs: list | None = None,
                 outputs: list | None = None, params: dict | None = None) -> None:
        payload = {"seq": len(self.steps) + 1, "action": action,
                   "inputs": inputs or [], "outputs": outputs or [], "params": params or {}}
        prev = self.steps[-1]["step_hash"] if self.steps else ZERO
        self.steps.append({**payload, "prev_hash": prev, "step_hash": _digest(payload, prev)})

    def to_dict(self) -> dict:
        return {"run_id": self.run_id, "steps": self.steps,
                "chain_hash": self.steps[-1]["step_hash"] if self.steps else ZERO}


def verify_chain(chain: dict) -> bool:
    prev = ZERO
    for step in chain["steps"]:
        payload = {k: step[k] for k in ("seq", "action", "inputs", "outputs", "params")}
        if step["prev_hash"] != prev or _digest(payload, prev) != step["step_hash"]:
            return False
        prev = step["step_hash"]
    return chain.get("chain_hash") == prev
```

"""订单视图装载：三本账 + 底座库 → 按订单号索引的只读内存视图（World）。"""
from __future__ import annotations

import csv
import sqlite3
from pathlib import Path

from src.engine.audit import sha256_of
from src.synth.build_dataset import NATURAL_MISMATCH_SQL


def _read_csv(path: Path) -> list[dict]:
    if not path.is_file():
        raise FileNotFoundError(f"缺少数据文件：{path}（先跑 scripts/build_synthetic.py）")
    with open(path, newline="", encoding="utf-8") as f:
        return list(csv.DictReader(f))


def _group(rows: list[dict], key: str) -> dict[str, list[dict]]:
    out: dict[str, list[dict]] = {}
    for r in rows:
        out.setdefault(r[key], []).append(r)
    return out


class World:
    """一次引擎运行的全部输入。只读；分类器不得修改其中任何字段。"""

    def __init__(self, orders, items_by_order, bills_by_order, payments_by_order,
                 refunds_by_order, natural_ids, sources):
        self.orders = orders
        self.items_by_order = items_by_order
        self.bills_by_order = bills_by_order
        self.payments_by_order = payments_by_order
        self.refunds_by_order = refunds_by_order
        self.natural_ids = natural_ids
        self.sources = sources


def load_world(synthetic_dir: Path, db_path: Path) -> World:
    dirty = Path(synthetic_dir) / "dirty"
    bills = _read_csv(dirty / "platform_bills.csv")
    payments = _read_csv(dirty / "order_payments.csv")
    refunds = _read_csv(dirty / "refunds.csv")
    if not Path(db_path).is_file():
        raise FileNotFoundError(f"缺少底座数据库：{db_path}（先跑 scripts/build_db.py）")
    conn = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
    conn.row_factory = sqlite3.Row
    try:
        orders = {r["order_id"]: dict(r) for r in conn.execute(
            "SELECT order_id, order_status, order_approved_at, order_purchase_timestamp FROM orders")}
        items_by_order = _group([dict(r) for r in conn.execute(
            "SELECT order_id, order_item_id, product_id, seller_id, price, freight_value"
            " FROM order_items")], "order_id")
        natural_ids = {r["order_id"] for r in conn.execute(NATURAL_MISMATCH_SQL)}
    finally:
        conn.close()
    sources = {}
    for name, path, rows in (("platform_bills", dirty / "platform_bills.csv", bills),
                             ("order_payments", dirty / "order_payments.csv", payments),
                             ("refunds", dirty / "refunds.csv", refunds)):
        sources[name] = {"path": str(path), "sha256": sha256_of(path), "rows": len(rows)}
    return World(orders, items_by_order, _group(bills, "order_id"),
                 _group(payments, "order_id"), _group(refunds, "order_id"),
                 natural_ids, sources)
```

"""规则树：把订单视图判成六类发现。先形态后金额，一单至多一条金额类发现。

判定谓词与 P1 埋雷逐条对齐（spec 2026-09-26-p2-engine-design.md 第 1 节）；
金额期望复用 src/synth/bills.py 的规则卡实现——造数与判卷同一口径。
"""
from __future__ import annotations

from src.synth.bills import amounts_from_items
from src.synth.config import NO_BILL_STATUSES, TOLERANCE, WRONG_RATE


def classify(world) -> list[dict]:
    findings: list[dict] = []

    def emit(order_id, diff_type, variant, target_side, confidence, description, evidence):
        findings.append({"finding_id": "", "order_id": order_id, "diff_type": diff_type,
                         "variant": variant, "target_side": target_side,
                         "confidence": confidence, "description": description,
                         "evidence": evidence, "verified": False})

    # #1 孤儿账单：账单的订单号不在订单表
    for oid, bills in world.bills_by_order.items():
        if oid not in world.orders:
            emit(oid, "单边账", "孤儿账单", "bills", 1.0, "孤儿账单：订单不存在",
                 {"bill_ids": [b["bill_id"] for b in bills]})

    for oid, order in world.orders.items():
        items = world.items_by_order.get(oid, [])
        bills = world.bills_by_order.get(oid, [])
        pays = world.payments_by_order.get(oid, [])
        refunds = world.refunds_by_order.get(oid, [])
        paid = round(sum(float(p["payment_value"]) for p in pays), 2)
        approved = bool((order.get("order_approved_at") or "").strip())
        no_bill_status = order["order_status"] in NO_BILL_STATUSES

        # 退款侧异常：不该有的退款（活跃单、或未审批取消单出现退款）
        if refunds and not (no_bill_status and approved):
            emit(oid, "金额不符", "不该有的退款", "refunds", 0.8,
                 "非「已审批的取消单」出现退款记录",
                 {"refund_ids": [r["refund_id"] for r in refunds]})

        if no_bill_status:
            if bills:                                   # #2 状态不一致
                emit(oid, "状态不一致", "取消单被开账单", "bills", 1.0,
                     "取消/不可达订单出现在账单中",
                     {"bill_ids": [b["bill_id"] for b in bills]})
            if approved and pays:                       # #3 退款遗漏 / 退款金额不符
                if not refunds:
                    emit(oid, "退款遗漏", "退款缺失", "refunds", 1.0,
                         "已付款的取消单无退款记录", {"paid": paid})
                else:
                    for r in refunds:
                        if abs(float(r["refund_amount"]) - paid) > TOLERANCE:
                            emit(oid, "金额不符", "退款金额不符", "refunds", 0.9,
                                 "退款金额 ≠ 已付金额",
                                 {"refund_id": r["refund_id"], "paid": paid,
                                  "refund_amount": float(r["refund_amount"])})
            continue                                    # 取消单不做账单金额核对

        if not items:
            continue                                    # 无明细订单无账单期望（P0：775 单）

        if not bills:
            if pays:                                    # #4 删账单
                emit(oid, "单边账", "删账单", "bills", 1.0,
                     "账单缺失：有明细有支付但无账单", {"paid": paid})
            else:                                       # #5 双缺失
                emit(oid, "单边账", "账与支付均缺失", "bills", 0.8,
                     "账单与支付双双缺失", {"n_items": len(items)})
            continue

        if len({b["version"] for b in bills}) > 1:      # #6 版本差异（命中即停）
            emit(oid, "版本差异", "两版并存", "bills", 1.0, "两版账单并存，按最高版本对账",
                 {"bill_ids": [b["bill_id"] for b in bills],
                  "versions": sorted({b["version"] for b in bills})})
            continue
        if len(bills) > 1:                              # #7 重复结算
            emit(oid, "重复结算", "整单重复", "bills", 1.0, "同订单同版本账单重复",
                 {"bill_ids": [b["bill_id"] for b in bills]})
            continue
        if not pays:                                    # #8 删支付
            emit(oid, "单边账", "删支付", "payments", 1.0, "支付缺失",
                 {"bill_ids": [b["bill_id"] for b in bills]})
            continue

        # #9 账单侧金额 vs 规则卡期望（依次：货款→佣金→运费→结算额）
        exp_price, exp_freight, exp_fee = amounts_from_items(items)
        exp_settle = round(exp_price + exp_freight - exp_fee, 2)
        b = bills[0]
        ev = {"bill_id": b["bill_id"], "rate": b["commission_rate"]}
        if abs(float(b["price_amount"]) - exp_price) > TOLERANCE:
            emit(oid, "金额不符", "货款不符", "bills", 0.8, "账单货款 ≠ 明细货款合计",
                 {**ev, "expected": exp_price, "actual": float(b["price_amount"])})
        elif abs(float(b["commission_fee"]) - exp_fee) > TOLERANCE:
            if b["commission_rate"] == f"{WRONG_RATE:.2f}":
                emit(oid, "金额不符", "费率错误", "bills", 1.0,
                     f"佣金费率错误：{b['commission_rate']} ≠ 标准费率",
                     {**ev, "expected_fee": exp_fee, "actual_fee": float(b["commission_fee"])})
            else:
                emit(oid, "金额不符", "费率异常", "bills", 0.7,
                     f"佣金费率异常：{b['commission_rate']}（标准 0.20）",
                     {**ev, "expected_fee": exp_fee, "actual_fee": float(b["commission_fee"])})
        elif abs(float(b["freight_amount"]) - exp_freight) > TOLERANCE:
            if float(b["freight_amount"]) == 0:
                emit(oid, "金额不符", "漏运费", "bills", 1.0, "账单漏加运费",
                     {**ev, "expected_freight": exp_freight, "actual_freight": 0.0})
            else:
                emit(oid, "金额不符", "运费不符", "bills", 0.7, "账单运费 ≠ 明细运费合计",
                     {**ev, "expected_freight": exp_freight,
                      "actual_freight": float(b["freight_amount"])})
        elif abs(float(b["settlement_amount"]) - exp_settle) > TOLERANCE:
            emit(oid, "金额不符", "结算额不符", "bills", 0.8, "应付卖家 ≠ 规则卡期望",
                 {**ev, "expected": exp_settle, "actual": float(b["settlement_amount"])})
        elif abs(paid - round(exp_price + exp_freight, 2)) > TOLERANCE:
            # #10 账单侧干净 → 支付侧（自然差异的主要落点）
            emit(oid, "金额不符", "支付与单据不符", "payments", 0.9,
                 "支付合计 ≠ 明细货款+运费",
                 {"expected": round(exp_price + exp_freight, 2), "paid": paid})

    for i, f in enumerate(findings, 1):
        f["finding_id"] = f"F{i:06d}"
    return findings

"""验证器：对每条发现独立重推判定谓词。独立 = 不读 evidence，只凭类型从原始视图重算；
推得回去才 verified=True。结论落报告前必须过这道关（spec 第 2 节）。"""
from __future__ import annotations

from src.synth.bills import amounts_from_items
from src.synth.config import NO_BILL_STATUSES, TOLERANCE, WRONG_RATE


def _predicate(world, f: dict) -> bool:
    oid = f["order_id"]
    bills = world.bills_by_order.get(oid, [])
    pays = world.payments_by_order.get(oid, [])
    refunds = world.refunds_by_order.get(oid, [])
    order = world.orders.get(oid)
    if f["diff_type"] == "单边账" and f["variant"] == "孤儿账单":
        return order is None and len(bills) > 0
    if order is None:
        return False
    items = world.items_by_order.get(oid, [])
    no_bill_status = order["order_status"] in NO_BILL_STATUSES
    approved = bool((order.get("order_approved_at") or "").strip())
    paid = round(sum(float(p["payment_value"]) for p in pays), 2)

    if f["diff_type"] == "状态不一致":
        return no_bill_status and len(bills) > 0
    if f["diff_type"] == "退款遗漏":
        return no_bill_status and approved and bool(pays) and not refunds
    if f["diff_type"] == "金额不符" and f["variant"] == "不该有的退款":
        return bool(refunds) and not (no_bill_status and approved)
    if f["diff_type"] == "金额不符" and f["variant"] == "退款金额不符":
        return (bool(refunds) and no_bill_status and approved
                and any(abs(float(r["refund_amount"]) - paid) > TOLERANCE for r in refunds))
    if f["diff_type"] == "单边账" and f["variant"] == "删账单":
        return bool(items) and bool(pays) and not bills and not no_bill_status
    if f["diff_type"] == "单边账" and f["variant"] == "账与支付均缺失":
        return bool(items) and not bills and not pays and not no_bill_status
    if f["diff_type"] == "版本差异":
        return len({b["version"] for b in bills}) > 1
    if f["diff_type"] == "重复结算":
        return len(bills) > 1 and len({b["version"] for b in bills}) == 1
    if f["diff_type"] == "单边账" and f["variant"] == "删支付":
        return bool(items) and bool(bills) and not pays and not no_bill_status
    if f["diff_type"] == "金额不符" and len(bills) == 1 and items:
        exp_price, exp_freight, exp_fee = amounts_from_items(items)
        b = bills[0]
        v = f["variant"]
        if v == "货款不符":
            return abs(float(b["price_amount"]) - exp_price) > TOLERANCE
        if v == "费率错误":
            return (b["commission_rate"] == f"{WRONG_RATE:.2f}"
                    and abs(float(b["commission_fee"]) - exp_fee) > TOLERANCE)
        if v == "费率异常":
            return (b["commission_rate"] != f"{WRONG_RATE:.2f}"
                    and abs(float(b["commission_fee"]) - exp_fee) > TOLERANCE)
        if v == "漏运费":
            return float(b["freight_amount"]) == 0 and abs(exp_freight) > TOLERANCE
        if v == "运费不符":
            return float(b["freight_amount"]) != 0 and abs(float(b["freight_amount"]) - exp_freight) > TOLERANCE
        if v == "结算额不符":
            return abs(float(b["settlement_amount"]) - round(exp_price + exp_freight - exp_fee, 2)) > TOLERANCE
        if v == "支付与单据不符":
            bill_clean = (abs(float(b["price_amount"]) - exp_price) <= TOLERANCE
                          and abs(float(b["commission_fee"]) - exp_fee) <= TOLERANCE
                          and abs(float(b["freight_amount"]) - exp_freight) <= TOLERANCE
                          and abs(float(b["settlement_amount"])
                                  - round(exp_price + exp_freight - exp_fee, 2)) <= TOLERANCE)
            return bill_clean and abs(paid - round(exp_price + exp_freight, 2)) > TOLERANCE
    return False


def verify(world, findings: list[dict]) -> None:
    for f in findings:
        f["verified"] = _predicate(world, f)

```python
"""判分：引擎发现 vs 标准答案。口径（injection-spec-v1）：
检出 TP = 命中 (order_id, diff_type)；漏检 FN = 答案有未报；误报 FP = 报了答案无，
FP 再按 P0 自然差异口径分桶（自然 / 其他）。"""
from __future__ import annotations

DIFF_TYPES = ["单边账", "金额不符", "状态不一致", "退款遗漏", "重复结算", "版本差异"]


def score(gt_rows: list[dict], findings: list[dict], natural_ids: set[str]) -> dict:
    gt_keys = {(g["order_id"], g["diff_type"]) for g in gt_rows}
    find_keys = {(f["order_id"], f["diff_type"]) for f in findings}
    tp_keys = gt_keys & find_keys
    fn_keys = gt_keys - find_keys
    fp_keys = find_keys - gt_keys
    by_type = {}
    for dtype in DIFF_TYPES:
        t_g = {k for k in gt_keys if k[1] == dtype}
        t_f = {k for k in find_keys if k[1] == dtype}
        tp = len(t_g & t_f)
        by_type[dtype] = {"tp": tp, "fn": len(t_g - t_f), "fp": len(t_f - t_g),
                          "recall": round(tp / len(t_g), 4) if t_g else None}
    fp_natural = sorted(o for o, _ in fp_keys if o in natural_ids)
    fp_other = sorted([list(k) for k in fp_keys if k[0] not in natural_ids])
    return {
        "gt_total": len(gt_keys), "finding_total": len(find_keys),
        "tp": len(tp_keys), "fn": len(fn_keys), "fp": len(fp_keys),
        "fp_natural": len(fp_natural), "fp_natural_ids": fp_natural,
        "fp_other": len(fp_other), "fp_other_keys": fp_other,
        "recall": round(len(tp_keys) / len(gt_keys), 4) if gt_keys else None,
        "precision_other": (round(len(tp_keys) / (len(tp_keys) + len(fp_other)), 4)
                            if (tp_keys or fp_other) else None),
        "by_type": by_type,
    }
```
